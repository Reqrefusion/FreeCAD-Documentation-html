// Concept Design: Ultrasonic Booster Cooling Sleeve

$fn = 60;

// Dimensions Concept (Adjust to your booster)

booster_od = 50; // Outer Diameter of Booster

wall_thick = 3; // Sleeve Wall Thickness

air_gap = 6; // Airflow Plenum Space

sleeve_h = 90; // Height of the sleeve

fan_size = 40; // 40mm Cooling Fan Mount

module booster_sleeve() {

    difference() {

        union() {

            // Main Outer Shell

            cylinder(d = booster_od + (air_gap + wall_thick)*2, h = sleeve_h, center = true);

           

            // Fan Mount Block (Side Attachment)

            translate([booster_od/2 + air_gap + wall_thick, 0, 0])

                rotate([0, 90, 0])

                cube([fan_size + 6, fan_size + 6, 12], center = true);

        }

       

        // Inner Cutout for Booster (Leaves Air Gap)

        cylinder(d = booster_od + (air_gap*2), h = sleeve_h + 2, center = true);

       

        // Top & Bottom Clamping/Centering Lips

        translate([0,0, sleeve_h/2 - 4])

            cylinder(d1 = booster_od + (air_gap*2), d2 = booster_od, h = 4);

        translate([0,0, -sleeve_h/2])

            cylinder(d1 = booster_od, d2 = booster_od + (air_gap*2), h = 4);

        // Fan Air Inlet Hole

        translate([booster_od/2, 0, 0])

            rotate([0, 90, 0])

            cylinder(d = fan_size - 4, h = 30, center = true);

        // Exhaust Vents (Opposite Side)

        for (z = [-25 : 15 : 25]) {

            translate([-(booster_od/2 + air_gap + wall_thick), 0, z])

                rotate([0, 90, 0])

                cube([8, 25, 20], center = true);

        }

       

        // Two-Piece Split Cut (Cut in half for easy mounting)

        // Uncomment below to inspect split line

        // translate([0, 50, 0]) cube([150, 100, sleeve_h + 10], center = true);

    }

}
