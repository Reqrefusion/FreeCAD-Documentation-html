// Parametric shallow box / tray

// Pocket: 175 × 85 × 3 mm deep

// Base: 3 mm thick

// Walls: 3 mm all around

// ===== Parameters =====

pocket_length = 175; // mm

pocket_width = 85; // mm

pocket_depth = 3; // mm

wall_thickness = 3; // mm

base_thickness = 3; // mm

// Derived dimensions

outer_length = pocket_length + 2 * wall_thickness;

outer_width = pocket_width + 2 * wall_thickness;

total_height = base_thickness + pocket_depth;

// ===== Model =====

difference() {

    // Outer solid

    cube([outer_length, outer_width, total_height]);

    // Pocket cavity

    translate([wall_thickness, wall_thickness, base_thickness])

        cube([pocket_length, pocket_width, pocket_depth + 0.1]); // +0.1 for clean cut

}
