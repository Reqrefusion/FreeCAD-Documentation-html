  1. ============================================================

  3. Units: millimeters
  4. ============================================================
  1. ---------- MAIN DIMENSIONS ----------

L = 220.0 # Overall mounting plate length H = 32.0 # Mounting plate height T = 5.0 # Mounting plate thickness R = 4.0 # Plate edge/end radius

  1. ---------- MOUNTING HOLES ----------

HOLE_D = 6.0 # Hole diameter HOLE_OFFSET_X = 18.0 # Hole distance from ends HOLE_Z = H / 2.0 # Vertical hole position

  1. ---------- HOOK PARAMETERS ----------

HOOK_COUNT = 4 HOOK_WIDTH = 9.0 # Width of each hook HOOK_THICK = 4.0 # Thickness of hook material HOOK_RADIUS = 4.0 # Sweep corner/profile radius

HOOK_START_Z = 11.0 # Hook attachment height HOOK_FORWARD = 42.0 # How far hook projects from plate HOOK_DROP = 15.0 # Downward curve HOOK_TIP_UP = 9.0 # Upward tip

  1. ---------- SPACING ----------

END_MARGIN = 28.0 SPACING = (L - 2 * END_MARGIN) / (HOOK_COUNT - 1)

plate_main = Part.makeBox(

right_end = Part.makeCylinder( 

   R,
   T,
   App.Vector(L - R, 0, H / 2),
   App.Vector(0, 1, 0)

)

center_plate = Part.makeBox( 

   L - 2 * R,
   T,
   H,
   App.Vector(R, 0, 0)

)

plate = center_plate.fuse(left_end).fuse(right_end)

   L - HOLE_OFFSET_X

  2. FUNCTION: CREATE ONE HOOK
  3. ============================================================

def create_hook(x_position, index): 

   # Front surface of mounting plate
   y0 = T

   # Path points
   #
   # Starts from mounting plate
   # moves outward
   # curves downward
   # then rises at the tip

   p1 = App.Vector(x_position, y0, HOOK_START_Z)

   p2 = App.Vector(
       x_position,
       y0 + 12,
       HOOK_START_Z
   )

   p3 = App.Vector(
       x_position,
       y0 + 25,
       HOOK_START_Z - 5
   )

   p4 = App.Vector(
       x_position,
       y0 + HOOK_FORWARD * 0.65,
       HOOK_START_Z - HOOK_DROP
   )

   p5 = App.Vector(
       x_position,
       y0 + HOOK_FORWARD * 0.85,
       HOOK_START_Z - HOOK_DROP + 3
   )

   p6 = App.Vector(
       x_position,
       y0 + HOOK_FORWARD,
       HOOK_START_Z - HOOK_DROP + HOOK_TIP_UP
   )

   points = [p1, p2, p3, p4, p5, p6]

   # Smooth B-Spline centerline
   curve = Part.BSplineCurve()
   curve.interpolate(points)

   path_edge = curve.toShape()
   path_wire = Part.Wire(path_edge)

   # Rectangular profile
   # Profile is perpendicular to initial Y-direction

   x1 = x_position - HOOK_WIDTH / 2.0
   x2 = x_position + HOOK_WIDTH / 2.0

   z1 = HOOK_START_Z - HOOK_THICK / 2.0
   z2 = HOOK_START_Z + HOOK_THICK / 2.0

   y = y0

   profile_edges = [
       Part.makeLine(
           App.Vector(x1, y, z1),
           App.Vector(x2, y, z1)
       ),
       Part.makeLine(
           App.Vector(x2, y, z1),
           App.Vector(x2, y, z2)
       ),
       Part.makeLine(
           App.Vector(x2, y, z2),
           App.Vector(x1, y, z2)
       ),
       Part.makeLine(
           App.Vector(x1, y, z2),
           App.Vector(x1, y, z1)
       )
   ]

   profile_wire = Part.Wire(profile_edges)

   # Sweep profile along path
   try:
       hook = path_wire.makePipeShell(
           [profile_wire],
           True,
           False
       )
   except:
       # Fallback
       hook = Part.makeCylinder(
           HOOK_THICK / 2,
           HOOK_FORWARD,
           App.Vector(x_position, y0, HOOK_START_Z),
           App.Vector(0, 1, 0)
       )

   # Hook object
   hook_obj = doc.addObject(
       "PartDesign::Feature",
       "Hook_%d" % index
   )

   hook_obj.Label = "Hook %d" % index
   hook_obj.Shape = hook
   hook_obj.ViewObject.ShapeColor = COPPER_COLOR

   return hook
  1. ============================================================
  2. CREATE FOUR HOOKS
  3. ============================================================

hooks = []

for i in range(HOOK_COUNT): 

   x = END_MARGIN + i * SPACING

   hook = create_hook(x, i + 1)

   hooks.append(hook)

combined = all_shapes[0]
